cmake_minimum_required(VERSION 3.21)
project(StayAliveMC VERSION 2.0.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "StayAliveMC requires Windows and the MSVC toolchain.")
endif()
enable_language(RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(stayalive_core STATIC core/package_debug.cpp core/startup.cpp)
target_include_directories(stayalive_core PUBLIC core)
target_compile_definitions(stayalive_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(stayalive_core PUBLIC ole32 shell32 uuid advapi32)
target_compile_options(stayalive_core PRIVATE /W4 /WX /utf-8 /permissive-)

add_executable(StayAliveMC WIN32 core/main.cpp core/app.rc)
target_link_libraries(StayAliveMC PRIVATE stayalive_core comctl32)
target_compile_options(StayAliveMC PRIVATE /W4 /WX /utf-8 /permissive-)
target_link_options(StayAliveMC PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/core/app.manifest")

include(CTest)
if(BUILD_TESTING)
  add_executable(stayalive_tests tests/tests.cpp)
  target_link_libraries(stayalive_tests PRIVATE stayalive_core)
  target_compile_options(stayalive_tests PRIVATE /W4 /WX /utf-8 /permissive-)
  add_test(NAME stayalive_tests COMMAND stayalive_tests)
endif()
